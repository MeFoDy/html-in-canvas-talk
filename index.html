3. Загружаем элемент как текстуру

const size = canvas.getElementImageDefaultSize(ticket);
const w = Math.ceil(size.width), h = Math.ceil(size.height);

const target = gl.TEXTURE_2D;
gl.bindTexture(target, texture);
gl.texImage2D(target, 0, gl.RGBA8, w, h, 0,
  gl.RGBA, gl.UNSIGNED_BYTE, null);

const canvasTransform = new DOMMatrix();

4. Применяем магию WebGL

vec2 p = uv * paperSize;
vec2 edge = min(p, paperSize - p);
vec2 outside = max(max(controls.xy - p, p - controls.zw), 0.0);
float freePaper = smoothstep(4.0, 28.0, length(outside))
                * smoothstep(1.0, 13.0, min(edge.x, edge.y));

float phase = p.x * 0.030 + p.y * 0.014 - time * 2.3;
vec2 bend = vec2(2.8 * cos(phase), 6.5 * sin(phase)) * freePaper;
color = texture(ticket, uv + bend / paperSize);

5. Обновляем текстуру и кадр

let frame, deadline;

const draw = now => {
  try { effect.draw(now / 1000); }
  catch { return restore(); }
  frame = requestAnimationFrame(draw);
};

canvas.onpaint = () => {
  try { effect.updateTexture(); }
  catch { return restore(); }
  clearTimeout(deadline);
  frame ??= requestAnimationFrame(draw);
};
